```javascript
async function run(manifestPath) {
  const manifest = JSON.parse(await readFile(manifestPath, 'utf8'));
  const labels = Array.isArray(manifest.words) ? manifest.words.map(text => ({ text })) : [];
  const words = labels.map(({ text }) => text);
  if (!labels.length || labels.length > 64 || words.some(word => typeof word !== 'string' || !word.trim() || word.trim() !== word || word === '背景') ||
      new Set(words).size !== labels.length || !Array.isArray(manifest.samples) || !manifest.samples.length || manifest.samples.length > 3000) {
    throw Error('清单需要 1—64 个不同的动作词和 1—3000 条样本。');
  }
  const policy = JSON.parse(await readFile(resolve(root, 'shared/unified-training.json'), 'utf8'));
  const samples = [];
  for (const [index, entry] of manifest.samples.entries()) {
    samples.push(await sampleFromSelection(dirname(manifestPath), entry, labels, index));
  }
  const report = { status: '分区已检查', scope: '选段与分区演示', samples, checkedAt: new Date().toISOString() };
  try {
    const plan = splitJointTrainingPool(samples, labels, policy);  // [敲]
    report.samples = plan.samples;
    report.summary = plan.summary;
  } catch (error) {
    if (error.status !== 400) throw error;
    report.status = '分区未通过';
    report.reason = error.message;  // [敲]
  }
```

```python
def compute_stats(actions: list[dict[str, str]], tasks: list[dict[str, str]]) -> dict[str, Any]:
    """把工作簿行变成例会能直接读的数字，不把空状态算成完成。"""
    total = len(actions)                                                  # [敲]
    builtin = sum(1 for row in actions if row.get("内置识别") == "已有内置识别")  # [敲]
    new_targets = total - builtin                                         # [敲]
    animated = sum(1 for row in actions if str(row.get("数字人示范", "")).startswith("已有"))  # [敲]
    status = Counter(row.get("跟进状态", "未填") for row in actions)       # [敲]
    task_status = Counter(row.get("状态", "未填") for row in tasks)        # [敲]
    return {                                                              # [敲]
        "total": total,
        "builtin": builtin,
        "newTargets": new_targets,
        "animated": animated,
        "actionStatus": dict(status),
        "taskStatus": dict(task_status),
        "completionClaimAllowed": False,
        "note": "数字来自计划表；内置识别和已有示范都不等于连续段落验收。",
    }


def render_markdown(
    plan: dict[str, Any],
    stats: dict[str, Any],
    outputs: list[dict[str, Any]],
    health: list[dict[str, Any]] | None,
) -> str:
    """生成一页 Markdown 例会报告。"""
    lines = [                                                            # [敲]
        "# 语桥四岗项目例会 · 自动生成",
        "",
        f"- 生成时间：{time.strftime('%Y-%m-%d %H:%M:%S')}",
        f"- 项目计划更新：{plan.get('updated', '未知')}",
        f"- 当前下一步：{plan.get('nextCaption', '未填写')}",
        "",
        "## 医院互助 30 词计划",
        "",
        f"- 场景动作总数：{stats['total']}",                               # [敲]
        f"- 已有内置识别：{stats['builtin']}",                             # [敲]
        f"- 计划新增识别：{stats['newTargets']}",                          # [敲]
        f"- 已有数字人示范：{stats['animated']}（未专业验收）",             # [敲]
        f"- 跟进状态：{json.dumps(stats['actionStatus'], ensure_ascii=False)}",
        "",
        "## 四岗输出",
        "",
    ]                                                                    # [敲]
    for item in outputs:                                                 # [敲]
        mark = "存在" if item["exists"] else "缺失"
        lines.append(f"- {item['role']}：{mark}｜{item['path']}")         # [敲]
    if health:
        lines += ["", "## 健康检查", ""]
        for item in health:
            lines.append(f"- {item['role']}：{item['status']}（{item['url']}）")
    lines += [
        "",
        "## 边界",
        "",
        "- 计划数字不等于识别准确率。",                                    # [敲]
        "- 缺失输出如实报缺，不按“文件存在”推断完成。",                      # [敲]
        "- 医院互助只做沟通辅助，不替代诊断、挂号受理或医疗决策。",          # [敲]
    ]
    return "\n".join(lines) + "\n"                                      # [敲]
```
