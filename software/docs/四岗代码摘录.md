```python
def analyze_frame(source: str, args: argparse.Namespace, out_dir: Path) -> dict[str, Any]:
    """核心流程：取帧 → 提示词 → MiniCPM → JSON 解析 → 证据闸门。"""
    mode = "demo"                                        # [敲]
    image_path: Path | None = None
    capture_info: dict[str, Any] = {}

    if source == "--camera":                             # [敲]
        image_path, capture_info = capture_frame(args.camera, out_dir)
        mode = "local" if not args.endpoint else "endpoint"
    elif source == "--image":                            # [敲]
        image_path = Path(args.image).expanduser().resolve()
        if not image_path.exists():
            raise ObservationError(f"图片不存在：{image_path}")
        mode = "local" if not args.endpoint else "endpoint"

    prompt = build_prompt(args.scene)                    # [敲]

    if mode == "demo":                                   # [敲]
        raw = demo_raw()
    elif mode == "endpoint":                             # [敲]
        assert image_path is not None
        text = call_endpoint(args.endpoint, args.model, image_path, prompt, args.timeout)
        raw = parse_json_response(text)
    else:                                                # [敲]
        assert image_path is not None
        text = call_local(args.model, image_path, prompt)
        raw = parse_json_response(text)

    observation = normalize_observation(                 # [敲]
        raw,
        source=str(image_path) if image_path else "demo://hospital_mutual_aid",
        mode=mode,
    )
    if capture_info:
        observation["capture"] = capture_info

    if observation["dropped_candidates"]:                # [敲]
        print(f"[闸门] 丢弃 {len(observation['dropped_candidates'])} 条无证据候选")
    if not observation["sign_candidates"]:               # [敲]
        print("[闸门] 本次只输出视觉观察，不输出任何词")
    return observation
```

```python
def summarize_result(result, image_path: str, model_path: str) -> dict[str, Any]:
    """核心流程：逐实例取框、类别、mask → 统一 JSON。"""
    names = result.names or {}                                           # [敲]
    height, width = result.orig_shape[:2]                                # [敲]
    boxes = result.boxes                                               # [敲]
    masks = result.masks                                               # [敲]

    instances: list[dict[str, Any]] = []
    for index in range(len(boxes)):                                       # [敲]
        cls_id = int(boxes.cls[index].item())                             # [敲]
        confidence = float(boxes.conf[index].item())                      # [敲]
        xyxy = [round(float(v), 2) for v in boxes.xyxy[index].tolist()]   # [敲]
        instance = {
            "id": index + 1,
            "class": str(names.get(cls_id, cls_id)),
            "class_id": cls_id,
            "confidence": round(confidence, 4),
            "box": xyxy,
            "mask_area": None,
            "mask_polygon": [],
        }
        if masks is not None and masks.data is not None and index < len(masks.data):  # [敲]
            mask = masks.data[index].detach().cpu().numpy()                # [敲]
            instance["mask_area"] = int((mask > 0.5).sum())                # [敲]
            instance["mask_polygon"] = mask_to_polygon(mask, width, height)  # [敲]
        instances.append(instance)

    counts: dict[str, int] = {}                                            # [敲]
    for item in instances:                                                 # [敲]
        counts[item["class"]] = counts.get(item["class"], 0) + 1           # [敲]

    return {
        "schema_version": SCHEMA_VERSION,
        "module": MODULE_ID,
        "mode": "yolo_instance_segmentation",
        "source": str(image_path),
        "model": str(model_path),
        "image": {"width": width, "height": height},
        "instances": instances,
        "counts": counts,
        "notes": [
            "COCO 预训练类别不含 hand；手部区域需另接 MediaPipe 或自定义模型。",
            "mask 多边形为轮廓简化结果，不是像素级精确边界。",
            "分割结果不是手语词义，也不做身份识别。",
        ],
        "created_at": time.strftime("%Y-%m-%dT%H:%M:%S%z"),
    }


def run_yolo(model, image_path: str, conf: float = 0.25):
    """只推理一次，同时返回结构化结果和原始结果供标注图复用。"""
    results = model.predict(source=image_path, conf=conf, verbose=False)  # [敲]
    if not results:
        raise VisionError("YOLO 没有返回结果。")
    raw = results[0]                                                     # [敲]
    return summarize_result(raw, image_path, getattr(model, "ckpt_path", DEFAULT_MODEL)), raw
```

```python
async def run_hospital_workflow(
    image_bytes: bytes,
    vision_url: str = "",
    llm_url: str = "",
    timeout: float = DEFAULT_TIMEOUT,
) -> dict[str, Any]:
    """核心流程：并行调用 → 合并 → 策略闸门 → 统一结果。"""
    if not image_bytes:
        raise WorkflowError("图片为空。")
    if len(image_bytes) > 8 * 1024 * 1024:
        raise WorkflowError("图片超过 8MB。")

    workflow_id = uuid.uuid4().hex[:12]                                  # [敲]
    started = time.perf_counter()                                       # [敲]

    vision_task = call_vision(image_bytes, vision_url, timeout)          # [敲]
    llm_task = call_llm(image_bytes, llm_url, timeout)                   # [敲]
    try:
        vision, observation = await asyncio.wait_for(                       # [敲]
            asyncio.gather(vision_task, llm_task), timeout=timeout          # [敲]
        )
    except asyncio.TimeoutError as exc:                                  # [敲]
        raise WorkflowError(f"工作流超时（>{timeout:.0f}s）。") from exc    # [敲]

    candidates, dropped = [], []                                         # [敲]
    for item in observation.get("sign_candidates", []) or []:           # [敲]
        word = str(item.get("word", "")).strip()
        evidence = str(item.get("evidence", "")).strip()
        if not word or not evidence:                                     # [敲]
            dropped.append({"word": word, "reason": "no_evidence"})      # [敲]
            continue
        candidates.append({                                              # [敲]
            "word": word,
            "evidence": evidence,
            "confidence": max(0.0, min(1.0, float(item.get("confidence", 0) or 0))),
            "status": "candidate_requires_human_review",
        })

    if observation.get("medical_advice"):                                # [敲]
        dropped.append({"word": None, "reason": "medical_advice_blocked"})  # [敲]

    merged = {                                                           # [敲]
        "scene": "hospital_mutual_aid",
        "instances": vision.get("instances", []),
        "counts": vision.get("counts", {}),
        "observed": observation.get("observed", []),
        "unknowns": observation.get("unknowns", []),
        "sign_candidates": candidates,
    }
    result = {                                                           # [敲]
        "schema_version": SCHEMA_VERSION,
        "module": MODULE_ID,
        "workflow_id": workflow_id,
        "status": "completed",
        "result": merged,
        "policy": {
            "medical_advice": False,
            "human_review_required": True,
            "dropped_candidates": dropped,
            "no_evidence_no_word": True,
        },
        "timings_ms": {"total": round((time.perf_counter() - started) * 1000, 1)},
        "created_at": now(),
    }
    JOBS[workflow_id] = {"status": "completed", "result": result}       # [敲]
    return result
```

```python
def compute_stats(actions: list[dict[str, str]], tasks: list[dict[str, str]]) -> dict[str, Any]:
    """把工作簿行变成例会能直接读的数字，不把空状态算成完成。"""
    total = len(actions)                                                  # [敲]
    builtin = sum(1 for row in actions if row.get("内置识别") == "已有内置识别")  # [敲]
    new_targets = total - builtin                                         # [敲]
    animated = sum(1 for row in actions if str(row.get("数字人示范", "")).startswith("已有"))  # [敲]
    status = Counter(row.get("跟进状态", "未填") for row in actions)       # [敲]
    task_status = Counter(row.get("状态", "未填") for row in tasks)        # [敲]
    return {                                                              # [敲]
        "total": total,
        "builtin": builtin,
        "newTargets": new_targets,
        "animated": animated,
        "actionStatus": dict(status),
        "taskStatus": dict(task_status),
        "completionClaimAllowed": False,
        "note": "数字来自计划表；内置识别和已有示范都不等于连续段落验收。",
    }


def render_markdown(
    plan: dict[str, Any],
    stats: dict[str, Any],
    outputs: list[dict[str, Any]],
    health: list[dict[str, Any]] | None,
) -> str:
    """生成一页 Markdown 例会报告。"""
    lines = [                                                            # [敲]
        "# 语桥四岗项目例会 · 自动生成",
        "",
        f"- 生成时间：{time.strftime('%Y-%m-%d %H:%M:%S')}",
        f"- 项目计划更新：{plan.get('updated', '未知')}",
        f"- 当前下一步：{plan.get('nextCaption', '未填写')}",
        "",
        "## 医院互助 30 词计划",
        "",
        f"- 场景动作总数：{stats['total']}",                               # [敲]
        f"- 已有内置识别：{stats['builtin']}",                             # [敲]
        f"- 计划新增识别：{stats['newTargets']}",                          # [敲]
        f"- 已有数字人示范：{stats['animated']}（未专业验收）",             # [敲]
        f"- 跟进状态：{json.dumps(stats['actionStatus'], ensure_ascii=False)}",
        "",
        "## 四岗输出",
        "",
    ]                                                                    # [敲]
    for item in outputs:                                                 # [敲]
        mark = "存在" if item["exists"] else "缺失"
        lines.append(f"- {item['role']}：{mark}｜{item['path']}")         # [敲]
    if health:
        lines += ["", "## 健康检查", ""]
        for item in health:
            lines.append(f"- {item['role']}：{item['status']}（{item['url']}）")
    lines += [
        "",
        "## 边界",
        "",
        "- 计划数字不等于识别准确率。",                                    # [敲]
        "- 缺失输出如实报缺，不按“文件存在”推断完成。",                      # [敲]
        "- 医院互助只做沟通辅助，不替代诊断、挂号受理或医疗决策。",          # [敲]
    ]
    return "\n".join(lines) + "\n"                                      # [敲]
```
